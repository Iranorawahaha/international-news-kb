## 3️⃣ 精选规则（核心算法）

### 3.1 候选池：从三大看板抽取当日新闻
```python
# scripts/email/select_daily_brief.py
def select_today_news(date_str):
    """从三大看板抽取当日新闻，按"高/中/低"三档输出"""
    # 1. 加载三大数据源
    intl = load_archive('data/news-data.json')      # 国际
    ai = load_archive('data/ai-news.json')           # AI
    cn = load_archive('data/china-news.json')        # 国内

    # 2. 抽取"今天"的所有新闻
    candidates = []
    for source, archive in [('intl', intl), ('ai', ai), ('cn', cn)]:
        for art in archive.get(date_str, []):
            art['_source_board'] = source
            candidates.append(art)
    return candidates
```

### 3.2 评分公式（V2.0 体系对齐）
```python
def score_article(art, today_date):
    """为每条候选新闻计算"今日早报"得分"""
    score = 0

    # 1. 重要性等级（最重要 50 分）
    importance = get_importance(art)  # 复用 intl_template_v15.html 的 getImportance
    if importance['level'] == 'high': score += 50
    elif importance['level'] == 'medium': score += 25
    else: score += 10

    # 2. 时效性（今天 +30 / 昨天 +10 / 更早 0）
    if art.get('date') == today_date: score += 30
    elif art.get('date') == yesterday_date: score += 10

    # 3. 官方信源加权（白宫/国务院等 +20）
    if art.get('is_official'): score += 20

    # 4. priority_score 加权（每 10 分 +5）
    score += (art.get('priority_score') or 0) // 10 * 5

    # 5. 跨看板交叉验证（被多个看板覆盖的话题 +15）
    if art.get('cross_ref_count', 0) >= 2: score += 15

    # 6. 标题包含"突发/首次/独家/重大"等信号（+10）
    if any(k in art.get('title_zh','') for k in ['突发','首次','独家','重大','紧急']):
        score += 10

    return score
```

### 3.3 精选规则
- **3-5 条深度解读**（红色高优）：score ≥ 80 + 重要性=高
- **5-7 条简报**（黄色中优）：50 ≤ score < 80
- **5 条一句话速览**：score 最高的 5 条 + 跨领域话题

### 3.4 多样性约束
- 同一栏目（中国/美国/欧洲等）最多 2 条深度解读
- 同一信源（路透/BBC 等）最多 3 条简报
- 确保至少 1 条官方信源、1 条中国相关、1 条美国相关、1 条 AI/科技

### 6.2 精选算法：`select_daily_brief.py`（核心）
```python
import json, sys
from pathlib import Path
from datetime import datetime, timedelta

sys.path.insert(0, '.')
from scripts.classify_columns import classify_column

PROJ = Path('/Users/xiaoxiao/WorkBuddy/2026-07-29-17-06-50')
DATA = PROJ / 'data'

HIGH_KEYS = ['中国', '习近平', '中美', '中俄', '台海', '涉华', '对华',
             'China', 'Chinese', 'Beijing', 'TikTok', '华为', 'USTR',
             '制裁', '实体清单', '出口管制', 'DeepSeek', 'Qwen',
             'Trump', '特朗普', '鲁比奥', 'Rubio', 'Bessent', '贝森特']
MED_KEYS = ['伊朗', 'Iran', 'Hormuz', '霍尔木兹', '俄乌', 'Ukraine', 'Russia',
            '朝鲜', 'North Korea', '金正恩', '以色列', 'Israel', '加沙',
            '莫迪', 'Modi', '泽连斯基', 'Zelensky', '普京', 'Putin']

def get_importance(art):
    text = (art.get('title_zh','') or art.get('title','') or '') + ' ' + (art.get('title_en','') or '')
    if any(k in text for k in HIGH_KEYS): return 'high'
    if any(k in text for k in MED_KEYS): return 'medium'
    return 'low'

def score(art, today):
    s = 0
    imp = get_importance(art)
    s += {'high': 50, 'medium': 25, 'low': 10}[imp]
    if art.get('date') == today: s += 30
    elif art.get('date') == (datetime.strptime(today,'%Y-%m-%d') - timedelta(days=1)).strftime('%Y-%m-%d'):
        s += 10
    if art.get('is_official'): s += 20
    s += (art.get('priority_score') or 0) // 10 * 5
    if any(k in (art.get('title_zh','') or '') for k in ['突发','首次','独家','重大']):
        s += 10
    return s

def select(today):
    """返回 3-5 条深度 + 5-7 条简报 + 5 条速览"""
    candidates = []
    for board, fname in [('intl','news-data.json'), ('ai','ai-news.json'), ('china','china-news.json')]:
        try:
            d = json.load(open(DATA / fname))
            for art in d.get('archive',{}).get(today, []):
                art['_board'] = board
                candidates.append(art)
        except FileNotFoundError:
            continue

    # 评分
    for a in candidates:
        a['_score'] = score(a, today)
        a['_imp'] = get_importance(a)

    # 深度解读：score >= 80 + 高优先级 + 多样性约束
    deep = [a for a in candidates if a['_score'] >= 80 and a['_imp'] == 'high']
    # 简报：50-80 分
    brief = [a for a in candidates if 50 <= a['_score'] < 80]
    # 速览：top 5（跨领域）
    brief_sorted = sorted(brief, key=lambda a: -a['_score'])[:5]

    # 多样性：深度解读每栏目最多 2 条
    by_col = {}
    deep_filtered = []
    for a in sorted(deep, key=lambda a: -a['_score']):
        col = a.get('column', '其他')
        by_col[col] = by_col.get(col, 0) + 1
        if by_col[col] <= 2 and len(deep_filtered) < 5:
            deep_filtered.append(a)

    return {
        'deep': deep_filtered,
        'brief': sorted(brief, key=lambda a: -a['_score'])[:7],
        'quick': brief_sorted,
    }
```

### 6.4 SMTP 发送：`send_email.py`
```python
import smtplib, ssl, yaml, sys
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText
from email.utils import formatdate

def send(html_path, config_path):
    cfg = yaml.safe_load(open(config_path))
    smtp = cfg['smtp']['primary']
    msg = MIMEMultipart('alternative')
    msg['Subject'] = f"📧 Ira 早报 · {datetime.now().strftime('%Y-%m-%d %a')}"
    msg['From'] = f"{smtp['from_name']} <{smtp['from_addr']}>"
    msg['To'] = ', '.join(s['email'] for s in cfg['subscribers'] if s['enabled'])
    msg['Date'] = formatdate(localtime=True)

    html = open(html_path, encoding='utf-8').read()
    msg.attach(MIMEText(html, 'html', 'utf-8'))

    ctx = ssl.create_default_context()
    with smtplib.SMTP_SSL(smtp['host'], smtp['port'], context=ctx) as s:
        s.login(smtp['username'], smtp['password'])
        s.send_message(msg)
    print(f"✅ 已发送给 {len([s for s in cfg['subscribers'] if s['enabled']])} 位订阅者")
```

## 8️⃣ 异常处理

| 场景 | 检测方法 | 处理策略 |
|------|----------|----------|
| 看板数据未更新 | `lastUpdated` 不含今天 | 跳过发送 + 飞书告警 |
| SMTP 连接失败 | 捕获 `smtplib.SMTPException` | 重试 1 次 → 切备用 SMTP → 飞书告警 |
| 模板渲染失败 | Jinja2 异常 | 飞书告警 + 保留旧模板兜底 |
| 收件人地址无效 | `smtplib.SMTPRecipientsRefused` | 自动退订 + 飞书通知 |
| 邮件被识别为垃圾邮件 | 退信率 > 20% | 检查主题/发件人/SPF |
| 磁盘空间不足 | 渲染前检查 `/tmp` 剩余 | 清理 7 天前旧邮件 → 重试 |

## 1️⃣1️⃣ 成本与风险评估

### 成本
- **QQ 邮箱 SMTP**：免费（每天 < 50 封不会触发限制）
- **模板渲染**：本地 Python，几乎无成本
- **WorkBuddy automation**：已包含在套餐内

### 风险
- ⚠️ **邮件进垃圾箱**：通过 SPF/DKIM 配置 + 避免营销词汇降低概率
- ⚠️ **授权码泄露**：配置文件 `.gitignore` + 系统 Keychain 存储
- ⚠️ **数据延迟**：早报发送依赖 06:00 数据刷新，建议加 buffer（07:30 足够）
